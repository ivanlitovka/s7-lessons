import pyspark
from pyspark.sql import SparkSession

spark = SparkSession.builder \
                    .master("local") \
                    .appName("Learning DataFrames") \
                    .getOrCreate()
events = spark.read.json("/user/master/data/events/date=2022-05-01").cache()


# 1. Фильтруем только сообщения, где получатель — физическое лицо (message_to не NULL)
window = Window().partitionBy('event.message_from').orderBy('event.message_ts')

# lag_7 = получатель 7 сообщений назад
dfWithLag = events.withColumn("lag_7", F.lag("event.message_to", 7).over(window))

# Вывод: две колонки, фильтр null, сортировка по message_to по убыванию
dfWithLag.select(
    F.col("event.message_from").alias("message_from"),
    F.col("lag_7")
) \
    .filter(F.col("lag_7").isNotNull()) \
    .orderBy(F.col("message_from").desc()) \
    .show(10, False)
